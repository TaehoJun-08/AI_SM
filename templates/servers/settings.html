📖 How the API works 
# send a message (with an embed) as your companion
curl -X POST {{ request.host_url }}api/bot/messages \
  -H "Authorization: Bot YOUR_TOKEN" -H "Content-Type: application/json" \
  -d '{"channel_id": 1, "content": "Hello!", "embed": {"title": "Status", "description": "All good", "color": "#10b981",
        "fields": [{"name": "Uptime", "value": "99.9%", "inline": true}]}}'

# register slash commands
curl -X PUT {{ request.host_url }}api/bot/commands -H "Authorization: Bot YOUR_TOKEN" \
  -H "Content-Type: application/json" -d '{"commands": [{"name": "ping", "description": "Say pong"}]}'

# when someone types /ping we POST this JSON to your Interaction URL:
# {"type":"command","command":"ping","args":"","user":{"id":3,"username":"maya_k"},"server_id":1,"channel_id":2,"bot_id":4}
# header X-Companion-Signature = sha256( sha256(token) + body ).  Reply with JSON {"content": "pong"} (or an embed)
# or post later through /api/bot/messages.

# a webhook only needs the secret URL
curl -X POST {{ request.host_url }}api/hooks/BOT_ID/TOKEN -H "Content-Type: application/json" -d '{"content": "Deployed 🚀"}'
